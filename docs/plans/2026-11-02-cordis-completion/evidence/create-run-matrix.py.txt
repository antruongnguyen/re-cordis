#!/usr/bin/env python3
"""Create S0-S2 shared acceptance. Run this saved script from the repository root."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import time

ROOT = Path.cwd().resolve()
OUT = ROOT / "target/create-20261104"
OUT.mkdir(parents=True, exist_ok=True)
results = []


def fingerprints():
    files = subprocess.check_output(
        ["git", "ls-files", "-z", "--cached", "--others", "--exclude-standard",
         "Cargo.toml", "Cargo.lock", "crates", "tests", "spikes/m0"], cwd=ROOT
    ).decode().split("\0")
    return {p: hashlib.sha256((ROOT / p).read_bytes()).hexdigest()
            for p in sorted(set(files)) if p and (ROOT / p).is_file()}


def run(label, command, env, timeout=600):
    start = time.monotonic()
    log = OUT / (label + ".log")
    with log.open("w") as stream:
        try:
            result = subprocess.run(command, cwd=ROOT, env=env, stdout=stream,
                                    stderr=subprocess.STDOUT, timeout=timeout)
            code = result.returncode
        except subprocess.TimeoutExpired:
            code = 124
            stream.write("\nAcceptance stage timed out.\n")
    record = dict(stage=label, command=command, exit=code,
                  seconds=round(time.monotonic() - start, 2), log=str(log))
    results.append(record)
    (OUT / "matrix.json").write_text(json.dumps(results, indent=2) + "\n")
    print(f"{label}: {'PASS' if code == 0 else 'FAIL'} ({record['seconds']}s)", flush=True)


before = fingerprints()
(OUT / "final-inputs.json").write_text(json.dumps(before, indent=2) + "\n")
tools = {}
for tc in ("stable", "1.85.0"):
    names = ("cargo", "rustc", "rustdoc", "cargo-clippy", "clippy-driver")
    paths = {name: subprocess.check_output(
        ["rustup", "which", "--toolchain", tc, name], text=True).strip()
        for name in names}
    bindir = str(Path(paths["rustc"]).parent)
    assert all(str(Path(p).parent) == bindir for p in paths.values())
    env = os.environ.copy()
    env.update(RUSTUP_TOOLCHAIN=tc, RUSTC=paths["rustc"], RUSTDOC=paths["rustdoc"],
               PATH=bindir + os.pathsep + env["PATH"], RUSTDOCFLAGS="-D warnings",
               CARGO_TARGET_DIR=str(ROOT / "target" / ("create-acceptance-" + tc)))
    versions = {name: subprocess.check_output([path, "--version"], env=env, text=True).strip()
                for name, path in paths.items()}
    tools[tc] = dict(paths=paths, versions=versions)
    (OUT / "tools.json").write_text(json.dumps(tools, indent=2) + "\n")
    cargo = paths["cargo"]
    common = ["--workspace", "--all-features", "--locked", "--offline"]
    stages = [
        ("check", [cargo, "check", *common, "--all-targets"]),
        ("test", [cargo, "test", *common]),
        ("clippy", [cargo, "clippy", *common, "--all-targets", "--", "-D", "warnings"]),
        ("rustdoc", [cargo, "doc", *common, "--no-deps"]),
        ("hmr-no-default", [cargo, "test", "-p", "re-cordis-hmr", "--no-default-features", "--locked", "--offline"]),
        ("m0-test", [cargo, "test", "--manifest-path", "spikes/m0/Cargo.toml", "--locked", "--offline"]),
        ("m0-clippy", [cargo, "clippy", "--manifest-path", "spikes/m0/Cargo.toml", "--all-targets", "--locked", "--offline", "--", "-D", "warnings"]),
    ]
    for example in ("lifecycle", "timer_lifecycle", "console", "boot_reconfigure", "config_file", "reload"):
        stages.append(("example-" + example, [cargo, "run", "--example", example,
                       "--all-features", "--locked", "--offline"]))
    if tc == "stable":
        stages += [("fmt", [cargo, "fmt", "--all", "--check"]),
                   ("m0-fmt", [cargo, "fmt", "--manifest-path", "spikes/m0/Cargo.toml", "--check"])]
    stages += [("docs", ["python3", "-B", "tests/docs/check.py", "--toolchain", tc]),
               ("loader-parity", ["python3", "-B", "tests/loader_parity/run.py", "--toolchain", tc])]
    if tc == "stable":
        stages += [("docs-regression", ["python3", "-B", "tests/docs/test_check.py"]),
                   ("core-parity", ["python3", "-B", "tests/parity/run.py"])]
    for label, command in stages:
        run(tc + "-" + label, command, env)

after = fingerprints()
summary = dict(passed=sum(r["exit"] == 0 for r in results), stages=len(results),
               input_count=len(before), inputs_unchanged=before == after,
               revision=subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip(),
               upstream_revision=subprocess.check_output(
                   ["git", "-C", "local/cordis", "rev-parse", "HEAD"], text=True).strip(),
               upstream_status=subprocess.check_output(
                   ["git", "-C", "local/cordis", "status", "--porcelain", "--untracked-files=no"], text=True))
(OUT / "final-summary.json").write_text(json.dumps(summary, indent=2) + "\n")
print(json.dumps(summary), flush=True)
raise SystemExit(0 if summary["passed"] == len(results) and before == after
                 and summary["upstream_revision"] == "35fdc969e4b6c48c48ed44717c7a018324e4dd88"
                 and not summary["upstream_status"] else 1)
